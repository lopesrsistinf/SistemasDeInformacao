#include <stdio.h>
#include <string.h>

int main() {
    int n, i, j;
    char valor[101]; 
    int leds[10] = {6, 2, 5, 5, 4, 5, 6, 3, 7, 6};

    if (scanf("%d", &n) == 1) {
        for (i = 0; i < n; i++) {
            scanf("%s", valor);
            
            long long total_leds = 0;
        
            for (j = 0; valor[j] != '\0'; j++) {
                int digito = valor[j] - '0';
                total_leds += leds[digito];
            }
            
            printf("%lld leds\n", total_leds);
        }
    }

    return 0;
}
